// Function to calculate EMA slope, manage arrays, and compute normalized percentage
calculate_ema_slope_percentage(_src, _length, _array_size) =>
    ema = ta.ema(_src, _length)
    slope = ema - ema[1]
    stdev_slope = ta.stdev(slope, 1000)
    normalized_slope = slope / stdev_slope
    
    // Create arrays for positive and negative slopes
    var float[] up_ema_slope = array.new_float(1, 0)
    var float[] down_ema_slope = array.new_float(1, 0)
    
    // Update arrays
    if normalized_slope > 0
        array.push(up_ema_slope, normalized_slope)
        if array.size(up_ema_slope) > _array_size
            array.shift(up_ema_slope)
    if normalized_slope < 0
        array.push(down_ema_slope, normalized_slope)
        if array.size(down_ema_slope) > _array_size
            array.shift(down_ema_slope)
    
    // Calculate averages
    up_avg = array.avg(up_ema_slope)
    down_avg = array.avg(down_ema_slope)
    
    // Normalize slope to percentage
    percentage = 0.5
    if (normalized_slope < up_avg and normalized_slope > down_avg)
        percentage := ((normalized_slope - down_avg) / (up_avg - down_avg))
    else
        if normalized_slope > up_avg
            percentage := 1
        else
            percentage := 0
    
    percentage

// Function to calculate PSAR
psar(start, inc, maximum, source) =>
    var float minTick = 0  // Minimum tick value
    var float out = na  // Output SAR
    var int pos = na  // Current position: 1 for uptrend, -1 for downtrend
    var float maxMin = na  // Maximum or Minimum Extreme Point
    var float acc = na  // Acceleration Factor
    var float prev = na  // Previous SAR value
    var bool outSet = false  // Flag to check if SAR is updated

    // Initialization
    if bar_index >= 1
        prev := out[1]
        outSet := false
        if bar_index == 1
            // Set initial values based on the trend
            if source > source[1]
                pos := 1
                maxMin := math.max(source, source[1])
                prev := math.min(source, source[1])
            else
                pos := -1
                maxMin := math.min(source, source[1])
                prev := math.max(source, source[1])
            
            acc := start
        else
            // Carry forward previous values
            pos := pos[1]
            acc := acc[1]
            maxMin := maxMin[1]

        // Update logic for uptrend
        if pos == 1
            if source > maxMin
                maxMin := source
                acc := math.min(acc + inc, maximum)
            
            if source <= prev
                pos := -1
                out := maxMin
                maxMin := source
                acc := start
                outSet := true
        else
            // Update logic for downtrend
            if source < maxMin
                maxMin := source
                acc := math.min(acc + inc, maximum)
            
            if source >= prev
                pos := 1
                out := maxMin
                maxMin := source
                acc := start
                outSet := true

        // Default SAR calculation
        if not outSet
            out := prev + acc * (maxMin - prev)

            if pos == 1 and out >= source
                out := source - minTick
            
            if pos == -1 and out <= source
                out := source + minTick

    out
